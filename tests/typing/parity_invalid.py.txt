    gi.require_versions({"Gst": 1})  # error
    GLib.strjoinv(",", (1, 2))  # error
    Gst.Buffer.new_wrapped(("a", "b"))  # error
    caps["name"]  # error
    structure[1] = 640  # error
    Gtk.Template()  # error
    Gtk.Template(filename=42)  # error
    Gtk.Template(string="x")(int)  # error
    Gtk.Template(string="x", filename="y")  # error
    events.GLibEventLoop(main_context="wrong")  # error
    GObject.Signal(flags="wrong")  # error
